cmake_minimum_required(VERSION 3.25)
project(SurgeNative LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use an x64 Microsoft C++ toolchain and Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SURGE_EXTRA_RANGE "Declare an additional owned code range for compatibility validation" OFF)
option(SURGE_ASAN "Instrument owned C++ targets for defensive qualification" OFF)
if(SURGE_ASAN)
  if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "ASan qualification requires Release")
  endif()
  # Confirm instrumentation in every C++ translation unit, including the collector.
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/fsanitize=address;/Zi;/FI${CMAKE_CURRENT_SOURCE_DIR}/asan_required.hpp>")
  add_link_options(/INCREMENTAL:NO /DEBUG)
  # Use the runtime paired with the selected compiler, not an unrelated PATH DLL.
  get_filename_component(SURGE_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  set(SURGE_ASAN_RUNTIME "${SURGE_COMPILER_DIR}/clang_rt.asan_dynamic-x86_64.dll")
  if(NOT EXISTS "${SURGE_ASAN_RUNTIME}")
    message(FATAL_ERROR "Selected MSVC toolchain has no ASan runtime")
  endif()
  configure_file("${SURGE_ASAN_RUNTIME}" "${CMAKE_BINARY_DIR}/clang_rt.asan_dynamic-x86_64.dll" COPYONLY)
  add_executable(asan_probe asan_probe.cpp)
  set_target_properties(asan_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
endif()
add_library(lab_module SHARED lab_module.cpp lab_code.asm lab_code.def)
add_library(lab_module_v2 SHARED lab_module.cpp lab_code_v2.asm lab_code.def)
# Exported begin/end markers describe actual code, not incremental-link jump thunks.
# Other linker layouts remain unsupported and are refused by the parser.
target_link_options(lab_module PRIVATE /INCREMENTAL:NO)
target_link_options(lab_module_v2 PRIVATE /INCREMENTAL:NO)
if(SURGE_EXTRA_RANGE)
  target_compile_definitions(lab_module PRIVATE SURGE_EXTRA_RANGE=1)
  target_compile_definitions(lab_module_v2 PRIVATE SURGE_EXTRA_RANGE=1)
endif()
add_library(surge_collector STATIC code_range_collector.cpp)
add_executable(surge_fixture fixture_runner.cpp)
add_executable(surge_owned_app owned_app.cpp)
add_dependencies(surge_owned_app lab_module lab_module_v2)
target_link_libraries(surge_owned_app PRIVATE surge_collector)
add_dependencies(surge_fixture lab_module lab_module_v2)
target_link_libraries(surge_collector PUBLIC bcrypt)
target_link_libraries(surge_fixture PRIVATE surge_collector)
add_executable(code_pe_tests code_pe_tests.cpp)
add_executable(manifest_pe_tests manifest_pe_tests.cpp)
add_executable(collector_smoke collector_smoke.cpp)
target_link_libraries(collector_smoke PRIVATE surge_collector)
add_dependencies(collector_smoke lab_module)
foreach(target lab_module lab_module_v2 surge_collector surge_fixture surge_owned_app code_pe_tests manifest_pe_tests collector_smoke)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/sdl;/guard:cf;/utf-8>)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  foreach(config RELEASE DEBUG MINSIZEREL RELWITHDEBINFO)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
  endforeach()
endforeach()
enable_testing()
add_test(NAME bounded_pe_parser COMMAND code_pe_tests)
add_test(NAME versioned_manifest_parser COMMAND manifest_pe_tests)

add_test(NAME collector_without_scenarios COMMAND collector_smoke)
